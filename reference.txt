script.js

const words = [
    { word: "elephant", hint: "A large mammal with a trunk" },
    { word: "pizza", hint: "Popular Italian food" },
    { word: "computer", hint: "Electronic device" },
    { word: "beach", hint: "Sandy shore near sea" }
];

const hangmanStages = [
``,
`
  |
  |
  |
__|__`,
`
 ______
  |
  |
  |
__|__`,
`
 ______
  |    O
  |
  |
__|__`,
`
 ______
  |    O
  |   /|\\
  |
__|__`,
`
 ______
  |    O
  |   /|\\
  |   / \\
__|__`
];

let selectedWord = "";
let hint = "";
let guessedLetters = [];
let wrongTries = 0;

function startGame() {
    const random = words[Math.floor(Math.random() * words.length)];
    selectedWord = random.word;
    hint = random.hint;

    guessedLetters = [];
    wrongTries = 0;

    document.getElementById("hintText").innerText = hint;
    document.getElementById("message").innerText = "";
    document.getElementById("letterInput").value = "";

    updateWord();
    updateHangman();
}

function updateWord() {
    let display = "";
    for (let letter of selectedWord) {
        display += guessedLetters.includes(letter) ? letter + " " : "_ ";
    }
    document.getElementById("wordDisplay").innerText = display.trim();
}

function updateHangman() {
    document.getElementById("hangmanArt").innerText =
        hangmanStages[wrongTries];
}

function guessLetter() {
    const input = document.getElementById("letterInput");
    let letter = input.value.toLowerCase();
    input.value = "";

    if (!letter.match(/[a-z]/)) {
        showMessage("Enter a valid letter", "red");
        return;
    }

    if (guessedLetters.includes(letter)) {
        showMessage("Already guessed!", "orange");
        return;
    }

    guessedLetters.push(letter);

    if (!selectedWord.includes(letter)) {
        wrongTries++;
        updateHangman();
    }

    updateWord();
    checkGameStatus();
}

function checkGameStatus() {
    if (wrongTries === hangmanStages.length - 1) {
        showMessage("❌ Game Over! Word: " + selectedWord, "red");
    } else if (
        selectedWord.split("").every(l => guessedLetters.includes(l))
    ) {
        showMessage("🎉 You Win!", "green");
    }
}

function showMessage(msg, color) {
    const message = document.getElementById("message");
    message.innerText = msg;
    message.style.color = color;
}

startGame();
